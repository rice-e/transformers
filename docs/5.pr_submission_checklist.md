def test_large_model_pt(self):

        "safe-text-generation",
        model="openai-community/gpt2",  # or another suitable model
        safety_level="moderate"
    )

    # Test that safety actually works with real toxicity
    prompt = "Tell me a nice story about"
    outputs = pipe(prompt, max_new_tokens=50, do_sample=False)

    # Verify output structure

    self.assertIn("generated_text", outputs[0])
    self.assertGreater(len(outputs[0]["generated_text"]), len(prompt))

    # Test with safety metadata
    outputs_meta = pipe(prompt, max_new_tokens=50, return_safety_metadata=True)
